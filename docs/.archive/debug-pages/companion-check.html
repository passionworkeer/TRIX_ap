ALTER TABLE profiles 
ADD COLUMN IF NOT EXISTS companion_id UUID 
REFERENCES profiles(id) ON DELETE SET NULL;

-- 添加索引
CREATE INDEX IF NOT EXISTS idx_profiles_companion_id 
ON profiles(companion_id);

-- 添加注释
